## 1. 模块总览

```

│   │   ├── commands/            # 【模块 1】Tauri 命令层

│   │   ├── repository/          # 【模块 4】数据访问层
│   │   ├── services/            # 【模块 5】服务层
│   │   └── utils/               # 【模块 6】工具库
│   └── Cargo.toml
└── src/                         # React 前端
    ├── components/              # 【模块 7】UI 组件

    ├── stores/                  # 【模块 9】状态管理
    └── utils/                   # 【模块 10】前端工具
```

#### 2.1 ClipboardManager (剪切板管理器)
```rust
// clipboard_manager.rs
pub struct ClipboardManager {
    repository: Arc<ClipboardRepository>,
    image_service: Arc<ImageService>,
    ocr_service: Arc<OCRService>,
    task_queue: Arc<AsyncTaskQueue>,
    platform_clipboard: Box<dyn PlatformClipboard>,
}

impl ClipboardManager {
    /// 启动剪切板监听
    pub async fn start_listening(&self) -> Result<()> {
        self.platform_clipboard.register_listener(|content| {
            self.handle_clipboard_change(content).await
        })
    }

    /// 处理剪切板变化
    async fn handle_clipboard_change(&self, content: ClipboardContent) -> Result<()> {
        // 1. 去重检查
        let hash = self.calculate_hash(&content);
        if self.repository.exists_by_hash(&hash).await? {
            return Ok(());
        }

        // 2. 上下文注入
        let metadata = self.platform_clipboard.get_active_window_info()?;

        // 3. 创建 Item
        let item = self.create_item(content, metadata)?;

        // 4. 同步保存
        self.repository.insert(&item).await?;

        // 5. 通知前端
        self.emit_event("clipboard:new", &item)?;

        // 6. 异步处理
        self.schedule_async_tasks(&item).await;

        Ok(())
    }

    /// 调度异步任务
    async fn schedule_async_tasks(&self, item: &ClipboardItem) {
        match item.type_.as_str() {
            "image" => {
                self.task_queue.spawn_thumbnail_task(item.id.clone()).await;
                self.task_queue.spawn_ocr_task(item.id.clone()).await;
            }
            "text" => {
                // 预留：AI 摘要任务
            }
            _ => {}
        }
    }
}
```

```rust
pub trait PlatformClipboard: Send + Sync {
    /// 注册剪切板监听器

    where
        F: Fn(ClipboardContent) -> BoxFuture<'static, Result<()>> + Send + 'static;
```

```rust
// linux.rs
#[cfg(target_os = "linux")]
pub struct LinuxClipboard {
    display_type: DisplayType,
    x11_clipboard: Option<X11Clipboard>,
    wayland_clipboard: Option<WaylandClipboard>,
}

enum DisplayType {
    X11,
    Wayland,
}

impl LinuxClipboard {
    pub fn new() -> Result<Self> {
        // 自动检测显示服务器
        let display_type = if std::env::var("WAYLAND_DISPLAY").is_ok() {
            DisplayType::Wayland
        } else {
            DisplayType::X11
        };

        match display_type {
            DisplayType::X11 => Self::init_x11(),
            DisplayType::Wayland => Self::init_wayland(),
        }
    }

    fn init_x11() -> Result<Self> {
        use x11_clipboard::Clipboard;
        let clipboard = Clipboard::new()?;
        Ok(Self {
            display_type: DisplayType::X11,
            x11_clipboard: Some(clipboard),
            wayland_clipboard: None,
        })
    }

    fn init_wayland() -> Result<Self> {
        // Wayland 实现
        todo!("Wayland clipboard support")
    }
}

impl PlatformClipboard for LinuxClipboard {
    fn register_listener<F>(&self, callback: F) -> Result<()>
    where
        F: Fn(ClipboardContent) -> BoxFuture<'static, Result<()>> + Send + 'static,
    {
        match self.display_type {
            DisplayType::X11 => {
                // X11 轮询方案（X11 无原生监听 API）
                let clipboard = self.x11_clipboard.as_ref().unwrap().clone();
                tokio::spawn(async move {
                    let mut last_content = String::new();
                    loop {
                        if let Ok(content) = clipboard.load_wait(...) {
                            if content != last_content {
                                callback(ClipboardContent::Text(content.clone())).await;
                                last_content = content;
                            }
                        }
                        tokio::time::sleep(Duration::from_millis(500)).await;
                    }
                });
                Ok(())
            }
            DisplayType::Wayland => {
                // Wayland 事件监听
                todo!()
            }
        }
    }

    fn get_active_window_info(&self) -> Result<WindowMetadata> {
        match self.display_type {
            DisplayType::X11 => {
                use x11rb::connection::Connection;
                use x11rb::protocol::xproto::*;

                let (conn, screen_num) = x11rb::connect(None)?;
                let screen = &conn.setup().roots[screen_num];

                // 获取活动窗口
                let active_window = conn.get_property(
                    false,
                    screen.root,
                    atom!("_NET_ACTIVE_WINDOW"),
                    AtomEnum::WINDOW,
                    0,
                    1,
                )?.reply()?;

                // 获取窗口标题
                let title = conn.get_property(
                    false,
                    window_id,
                    AtomEnum::WM_NAME,
                    AtomEnum::STRING,
                    0,
                    1024,
                )?.reply()?;

                Ok(WindowMetadata {
                    source_app: "Unknown".to_string(),  // X11 难以获取
                    window_title: String::from_utf8_lossy(&title.value).to_string(),
                    source_url: None,
                })
            }
            DisplayType::Wayland => todo!(),
        }
    }
}
```

```rust
impl PlatformClipboard for WindowsClipboard {

    where
        F: Fn(ClipboardContent) -> BoxFuture<'static, Result<()>> + Send + 'static,
    {
        unsafe {
            use windows::Win32::System::DataExchange::*;
            AddClipboardFormatListener(self.hwnd)?;
        }

        // 注册窗口消息处理
        // WM_CLIPBOARDUPDATE 消息触发 callback

        Ok(())
    }

    fn get_active_window_info(&self) -> Result<WindowMetadata> {
        unsafe {
            use windows::Win32::UI::WindowsAndMessaging::*;

            let hwnd = GetForegroundWindow();
            let mut title = vec![0u16; 256];
            GetWindowTextW(hwnd, &mut title);

            // 获取进程名
            let mut process_id = 0;
            GetWindowThreadProcessId(hwnd, Some(&mut process_id));
            let process_name = get_process_name(process_id)?;

            Ok(WindowMetadata {
                source_app: process_name,
                window_title: String::from_utf16_lossy(&title),
                source_url: None,
            })
        }
    }
}
```

#### 3.3 macOS 实现
```rust
// macos.rs
#[cfg(target_os = "macos")]
pub struct MacOSClipboard {
    pasteboard: NSPasteboard,
}

impl PlatformClipboard for MacOSClipboard {
    fn register_listener<F>(&self, callback: F) -> Result<()> {
        // 使用 NSPasteboard changeCount 轮询
        let pasteboard = self.pasteboard.clone();
        tokio::spawn(async move {
            let mut last_count = pasteboard.changeCount();
            loop {
                let current_count = pasteboard.changeCount();
                if current_count != last_count {
                    if let Ok(content) = pasteboard.stringForType(NSPasteboardTypeString) {
                        callback(ClipboardContent::Text(content)).await;
                    }
                    last_count = current_count;
                }
                tokio::time::sleep(Duration::from_millis(500)).await;
            }
        });
        Ok(())
    }

    fn get_active_window_info(&self) -> Result<WindowMetadata> {
        // 使用 NSWorkspace 获取活动应用
        unsafe {
            let workspace = NSWorkspace::sharedWorkspace();
            let active_app = workspace.frontmostApplication();

            Ok(WindowMetadata {
                source_app: active_app.localizedName().to_string(),
                window_title: "".to_string(),  // macOS 难以获取窗口标题
                source_url: None,
            })
        }
    }
}
```

**跨平台策略**:
- **统一接口**: 通过 `PlatformClipboard` trait 抽象平台差异
- **条件编译**: 使用 `#[cfg(target_os = "...")]` 编译对应平台代码
- **降级方案**: X11/macOS 使用轮询，Windows 使用原生事件

```rust
// clipboard_repository.rs
pub struct ClipboardRepository {
    pool: SqlitePool,
}

impl ClipboardRepository {
    /// 插入新项
    pub async fn insert(&self, item: &ClipboardItem) -> Result<()> {
        sqlx::query!(
            r#"
            INSERT INTO clipbox_items (
                id, type, content, original_content, metadata,
                tags, user_note, created_at, updated_at
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            "#,
            item.id,
            item.type_,
            item.content,
            item.original_content,
            item.metadata,
            item.tags,
            item.user_note,
            item.created_at,
            item.updated_at,
        )
        .execute(&self.pool)
        .await?;

        Ok(())
    }

    /// 混合搜索
    pub async fn search_multi_field(
        &self,
        query: &str,
        filters: SearchFilters,
    ) -> Result<Vec<ClipboardItem>> {
        let pattern = format!("%{}%", query);

        sqlx::query_as!(
            ClipboardItem,
            r#"
            SELECT * FROM clipbox_items
            WHERE is_deleted = 0
            AND (
                content LIKE ?1
                OR ocr_text LIKE ?1
                OR tags LIKE ?1
                OR user_note LIKE ?1
            )
            ORDER BY updated_at DESC
            LIMIT 50
            "#,
            pattern
        )
        .fetch_all(&self.pool)
        .await
    }

    /// 按 Hash 检查是否存在
    pub async fn exists_by_hash(&self, hash: &str) -> Result<bool> {
        let result = sqlx::query!(
            "SELECT COUNT(*) as count FROM clipbox_items WHERE content_hash = ?",
            hash
        )
        .fetch_one(&self.pool)
        .await?;

        Ok(result.count > 0)
    }
}
```

**跨平台考虑**: SQLite 天然跨平台，无需特殊处理。

```rust
pub struct ImageService {

impl ImageService {
    /// 保存原始图片
    pub async fn save_raw_image(&self, id: &str, data: &[u8]) -> Result<String> {
        let path = self.images_dir.join("raw").join(format!("{}.png", id));
        tokio::fs::write(&path, data).await?;
        Ok(path.to_string_lossy().to_string())
    }

    /// 生成缩略图
    pub async fn generate_thumbnail(&self, id: &str, raw_path: &str) -> Result<String> {
        let img = image::open(raw_path)?;
        let thumbnail = img.thumbnail(200, 200);

        let thumb_path = self.images_dir.join("thumbs").join(format!("{}_thumb.png", id));
        thumbnail.save(&thumb_path)?;

        Ok(thumb_path.to_string_lossy().to_string())
    }
}
```

#### 5.3 AsyncTaskQueue (异步任务队列)
```rust
// task_queue.rs
pub struct AsyncTaskQueue {
    tx: mpsc::Sender<Task>,
}

pub enum Task {
    GenerateThumbnail { item_id: String, raw_path: String },
    RunOCR { item_id: String, image_path: String },
    UpdateMetadata { item_id: String, key: String, value: String },
}

impl AsyncTaskQueue {
    pub fn new(repository: Arc<ClipboardRepository>) -> Self {
        let (tx, mut rx) = mpsc::channel::<Task>(100);

        // 启动后台任务处理器
        tokio::spawn(async move {
            while let Some(task) = rx.recv().await {
                match task {
                    Task::GenerateThumbnail { item_id, raw_path } => {
                        // 处理缩略图生成
                    }
                    Task::RunOCR { item_id, image_path } => {
                        // 处理 OCR
                    }
                    _ => {}
                }
            }
        });

        Self { tx }
    }

    pub async fn spawn_thumbnail_task(&self, item_id: String) {
        self.tx.send(Task::GenerateThumbnail { item_id, raw_path: "...".to_string() }).await;
    }
}
```

## 5. 跨平台模块总结

| 模块 | 跨平台策略 | 关键技术 |
|------|-----------|---------|
| Commands | 平台无关 | Tauri 命令 |
| Core | 平台无关 | 纯 Rust 逻辑 |
| Platform | **平台特定** | Trait + 条件编译 |
| Repository | 平台无关 | SQLite (跨平台) |
| Services | 平台无关 | `image` crate (纯 Rust) |
| Frontend | 平台无关 | React (Web 技术) |

**关键点**: 只有 Platform 层需要平台特定实现，其他模块完全跨平台。
