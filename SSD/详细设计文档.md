## 1. 核心数据模型

### 1.1 ClipboardItem (剪切板项)

```rust
// src-tauri/src/core/models.rs
use serde::{Deserialize, Serialize};
use sqlx::FromRow;

#[derive(Debug, Clone, Serialize, Deserialize, FromRow)]
pub struct ClipboardItem {
    /// 唯一标识符 (UUID v4)
    pub id: String,

    /// 类型: "text" | "image" | "file"
    #[sqlx(rename = "type")]
    pub type_: String,

    /// 内容 (可编辑)
    /// - text: 纯文本内容
    /// - image: null (内容在文件系统)
    /// - file: 文件路径
    pub content: Option<String>,

    /// 原始内容 (不可变备份)
    pub original_content: Option<String>,

    /// 内容 Hash (用于去重)
    pub content_hash: String,

    /// 图片路径 (JSON)
    /// 格式: {"raw": "/path/to/raw.png", "thumb": "/path/to/thumb.png"}
    pub image_paths: Option<String>,

    /// OCR 识别文本 (异步填充)
    pub ocr_text: Option<String>,

    /// 向量嵌入 (预留)
    pub embedding: Option<Vec<u8>>,

    /// 标签 (JSON 数组)
    /// 格式: ["#Work", "#Important"]
    pub tags: Option<String>,

    /// 用户备注
    pub user_note: Option<String>,

    /// 元数据 (JSON)
    /// 格式: {"source_app": "Chrome", "window_title": "...", "url": "..."}
    pub metadata: Option<String>,

    /// 是否为便签
    pub is_pinned: bool,

    /// 是否收藏
    pub is_favorite: bool,

    /// 软删除标记
    pub is_deleted: bool,

    /// 创建时间 (Unix 时间戳)
    pub created_at: i64,

    /// 更新时间 (Unix 时间戳，用于 LRU 排序)
    pub updated_at: i64,
}

impl ClipboardItem {
    /// 创建新的文本项
    pub fn new_text(content: String, metadata: WindowMetadata) -> Self {
        let hash = Self::calculate_hash(&content);
        let now = chrono::Utc::now().timestamp();

        Self {
            id: uuid::Uuid::new_v4().to_string(),
            type_: "text".to_string(),
            content: Some(content.clone()),
            original_content: Some(content),
            content_hash: hash,
            image_paths: None,
            ocr_text: None,
            embedding: None,
            tags: None,
            user_note: None,
            metadata: Some(serde_json::to_string(&metadata).unwrap()),
            is_pinned: false,
            is_favorite: false,
            is_deleted: false,
            created_at: now,
            updated_at: now,
        }
    }

    /// 创建新的图片项
    pub fn new_image(raw_path: String, thumb_path: String, metadata: WindowMetadata) -> Self {
        let hash = Self::calculate_hash(&raw_path);
        let now = chrono::Utc::now().timestamp();

        let image_paths = serde_json::json!({
            "raw": raw_path,
            "thumb": thumb_path,
        });

        Self {
            id: uuid::Uuid::new_v4().to_string(),
            type_: "image".to_string(),
            content: None,
            original_content: None,
            content_hash: hash,
            image_paths: Some(image_paths.to_string()),
            ocr_text: None,
            embedding: None,
            tags: None,
            user_note: None,
            metadata: Some(serde_json::to_string(&metadata).unwrap()),
            is_pinned: false,
            is_favorite: false,
            is_deleted: false,
            created_at: now,
            updated_at: now,
        }
    }

    /// 计算内容 Hash
    fn calculate_hash(content: &str) -> String {
        use sha2::{Sha256, Digest};
        let mut hasher = Sha256::new();
        hasher.update(content.as_bytes());
        format!("{:x}", hasher.finalize())
    }

    /// 获取图片路径
    pub fn get_image_paths(&self) -> Option<ImagePaths> {
        self.image_paths.as_ref().and_then(|json| {
            serde_json::from_str(json).ok()
        })
    }

    /// 获取元数据
    pub fn get_metadata(&self) -> Option<WindowMetadata> {
        self.metadata.as_ref().and_then(|json| {
            serde_json::from_str(json).ok()
        })
    }

    /// 获取标签列表
    pub fn get_tags(&self) -> Vec<String> {
        self.tags.as_ref()
            .and_then(|json| serde_json::from_str(json).ok())
            .unwrap_or_default()
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ImagePaths {
    pub raw: String,
    pub thumb: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct WindowMetadata {
    pub source_app: String,
    pub window_title: String,
    pub source_url: Option<String>,
}
```

```rust
#[derive(Debug, Clone)]
pub enum ClipboardContent {
    Text(String),
    Image(Vec<u8>),
    File(String),
}

impl ClipboardContent {
    pub fn get_type(&self) -> &str {
        match self {
            Self::Text(_) => "text",
            Self::Image(_) => "image",
            Self::File(_) => "file",
        }
    }

    pub fn to_string(&self) -> String {
        match self {
            Self::Text(s) => s.clone(),
            Self::Image(_) => "[Image]".to_string(),
            Self::File(path) => path.clone(),
        }
    }
}
```

### 2.1 PlatformClipboard Trait

```rust
// src-tauri/src/platform/clipboard_trait.rs
use async_trait::async_trait;
use std::future::Future;
use std::pin::Pin;

pub type BoxFuture<'a, T> = Pin<Box<dyn Future<Output = T> + Send + 'a>>;

#[async_trait]
pub trait PlatformClipboard: Send + Sync {
    /// 注册剪切板监听器
    /// 当剪切板内容变化时，调用 callback
    fn register_listener<F>(&self, callback: F) -> Result<(), ClipboardError>
    where
        F: Fn(ClipboardContent) -> BoxFuture<'static, Result<(), ClipboardError>> + Send + Sync + 'static;

    /// 获取当前剪切板内容
    async fn get_content(&self) -> Result<ClipboardContent, ClipboardError>;

    /// 设置剪切板内容
    async fn set_content(&self, content: ClipboardContent) -> Result<(), ClipboardError>;

    /// 获取活动窗口信息
    fn get_active_window_info(&self) -> Result<WindowMetadata, ClipboardError>;

    /// 清空剪切板
    async fn clear(&self) -> Result<(), ClipboardError>;
}

#[derive(Debug, thiserror::Error)]
pub enum ClipboardError {
    #[error("Failed to access clipboard: {0}")]
    AccessError(String),

    #[error("Unsupported content type")]
    UnsupportedType,

    #[error("Platform error: {0}")]
    PlatformError(String),
}
```

### 2.2 Linux 实现 (X11/Wayland)

```rust
// src-tauri/src/platform/linux.rs
#[cfg(target_os = "linux")]
pub mod linux {
    use super::*;
    use std::sync::{Arc, Mutex};
    use tokio::sync::mpsc;

    pub struct LinuxClipboard {
        display_type: DisplayType,
        last_content: Arc<Mutex<String>>,
        stop_signal: Arc<Mutex<bool>>,
    }

    enum DisplayType {
        X11,
        Wayland,
    }

    impl LinuxClipboard {
        pub fn new() -> Result<Self, ClipboardError> {
            // 检测显示服务器
            let display_type = if std::env::var("WAYLAND_DISPLAY").is_ok() {
                DisplayType::Wayland
            } else if std::env::var("DISPLAY").is_ok() {
                DisplayType::X11
            } else {
                return Err(ClipboardError::PlatformError(
                    "No display server detected".to_string()
                ));
            };

            Ok(Self {
                display_type,
                last_content: Arc::new(Mutex::new(String::new())),
                stop_signal: Arc::new(Mutex::new(false)),
            })
        }
    }

    #[async_trait]
    impl PlatformClipboard for LinuxClipboard {
        fn register_listener<F>(&self, callback: F) -> Result<(), ClipboardError>
        where
            F: Fn(ClipboardContent) -> BoxFuture<'static, Result<(), ClipboardError>> + Send + Sync + 'static,
        {
            match self.display_type {
                DisplayType::X11 => self.register_x11_listener(callback),
                DisplayType::Wayland => self.register_wayland_listener(callback),
            }
        }

        async fn get_content(&self) -> Result<ClipboardContent, ClipboardError> {
            match self.display_type {
                DisplayType::X11 => self.get_x11_content().await,
                DisplayType::Wayland => self.get_wayland_content().await,
            }
        }

        async fn set_content(&self, content: ClipboardContent) -> Result<(), ClipboardError> {
            use arboard::Clipboard;
            let mut clipboard = Clipboard::new()
                .map_err(|e| ClipboardError::AccessError(e.to_string()))?;

            match content {
                ClipboardContent::Text(text) => {
                    clipboard.set_text(text)
                        .map_err(|e| ClipboardError::AccessError(e.to_string()))?;
                }
                ClipboardContent::Image(data) => {
                    // 使用 arboard 设置图片
                    let img = image::load_from_memory(&data)
                        .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;
                    clipboard.set_image(arboard::ImageData {
                        width: img.width() as usize,
                        height: img.height() as usize,
                        bytes: img.to_rgba8().into_raw().into(),
                    })
                    .map_err(|e| ClipboardError::AccessError(e.to_string()))?;
                }
                _ => return Err(ClipboardError::UnsupportedType),
            }

            Ok(())
        }

        fn get_active_window_info(&self) -> Result<WindowMetadata, ClipboardError> {
            match self.display_type {
                DisplayType::X11 => self.get_x11_window_info(),
                DisplayType::Wayland => self.get_wayland_window_info(),
            }
        }

        async fn clear(&self) -> Result<(), ClipboardError> {
            self.set_content(ClipboardContent::Text(String::new())).await
        }
    }

    impl LinuxClipboard {
        /// X11 监听器实现 (轮询方案)
        fn register_x11_listener<F>(&self, callback: F) -> Result<(), ClipboardError>
        where
            F: Fn(ClipboardContent) -> BoxFuture<'static, Result<(), ClipboardError>> + Send + Sync + 'static,
        {
            let last_content = self.last_content.clone();
            let stop_signal = self.stop_signal.clone();

            tokio::spawn(async move {
                use arboard::Clipboard;
                let mut clipboard = Clipboard::new().expect("Failed to create clipboard");

                loop {
                    // 检查停止信号
                    if *stop_signal.lock().unwrap() {
                        break;
                    }

                    // 获取当前剪切板内容
                    if let Ok(text) = clipboard.get_text() {
                        let mut last = last_content.lock().unwrap();
                        if text != *last {
                            *last = text.clone();
                            drop(last);  // 释放锁

                            // 调用回调
                            let content = ClipboardContent::Text(text);
                            if let Err(e) = callback(content).await {
                                eprintln!("Clipboard callback error: {}", e);
                            }
                        }
                    }

                    // 轮询间隔 500ms
                    tokio::time::sleep(tokio::time::Duration::from_millis(500)).await;
                }
            });

            Ok(())
        }

        /// Wayland 监听器实现
        fn register_wayland_listener<F>(&self, callback: F) -> Result<(), ClipboardError>
        where
            F: Fn(ClipboardContent) -> BoxFuture<'static, Result<(), ClipboardError>> + Send + Sync + 'static,
        {
            // Wayland 同样使用轮询方案 (wl-clipboard 无原生事件)
            self.register_x11_listener(callback)
        }

        /// 获取 X11 剪切板内容
        async fn get_x11_content(&self) -> Result<ClipboardContent, ClipboardError> {
            use arboard::Clipboard;
            let mut clipboard = Clipboard::new()
                .map_err(|e| ClipboardError::AccessError(e.to_string()))?;

            // 优先尝试获取文本
            if let Ok(text) = clipboard.get_text() {
                return Ok(ClipboardContent::Text(text));
            }

            // 尝试获取图片
            if let Ok(img) = clipboard.get_image() {
                let rgba = image::RgbaImage::from_raw(
                    img.width as u32,
                    img.height as u32,
                    img.bytes.to_vec(),
                ).ok_or(ClipboardError::PlatformError("Invalid image data".to_string()))?;

                let mut buffer = Vec::new();
                rgba.write_to(&mut std::io::Cursor::new(&mut buffer), image::ImageFormat::Png)
                    .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;

                return Ok(ClipboardContent::Image(buffer));
            }

            Err(ClipboardError::UnsupportedType)
        }

        /// 获取 Wayland 剪切板内容
        async fn get_wayland_content(&self) -> Result<ClipboardContent, ClipboardError> {
            // 使用 arboard (支持 Wayland)
            self.get_x11_content().await
        }

        /// 获取 X11 活动窗口信息
        fn get_x11_window_info(&self) -> Result<WindowMetadata, ClipboardError> {
            use x11rb::connection::Connection;
            use x11rb::protocol::xproto::*;
            use x11rb::rust_connection::RustConnection;

            let (conn, screen_num) = RustConnection::connect(None)
                .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;

            let screen = &conn.setup().roots[screen_num];

            // 获取 _NET_ACTIVE_WINDOW
            let active_window_atom = conn.intern_atom(false, b"_NET_ACTIVE_WINDOW")
                .map_err(|e| ClipboardError::PlatformError(e.to_string()))?
                .reply()
                .map_err(|e| ClipboardError::PlatformError(e.to_string()))?
                .atom;

            let active_window_prop = conn.get_property(
                false,
                screen.root,
                active_window_atom,
                AtomEnum::WINDOW,
                0,
                1,
            )
            .map_err(|e| ClipboardError::PlatformError(e.to_string()))?
            .reply()
            .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;

            if active_window_prop.value.len() < 4 {
                return Ok(WindowMetadata {
                    source_app: "Unknown".to_string(),
                    window_title: "".to_string(),
                    source_url: None,
                });
            }

            let window_id = u32::from_ne_bytes([
                active_window_prop.value[0],
                active_window_prop.value[1],
                active_window_prop.value[2],
                active_window_prop.value[3],
            ]);

            // 获取窗口标题
            let title_prop = conn.get_property(
                false,
                window_id,
                AtomEnum::WM_NAME,
                AtomEnum::STRING,
                0,
                1024,
            )
            .map_err(|e| ClipboardError::PlatformError(e.to_string()))?
            .reply()
            .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;

            let window_title = String::from_utf8_lossy(&title_prop.value).to_string();

            // 获取 WM_CLASS (应用名称)
            let class_atom = conn.intern_atom(false, b"WM_CLASS")
                .map_err(|e| ClipboardError::PlatformError(e.to_string()))?
                .reply()
                .map_err(|e| ClipboardError::PlatformError(e.to_string()))?
                .atom;

            let class_prop = conn.get_property(
                false,
                window_id,
                class_atom,
                AtomEnum::STRING,
                0,
                1024,
            )
            .map_err(|e| ClipboardError::PlatformError(e.to_string()))?
            .reply()
            .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;

            let source_app = String::from_utf8_lossy(&class_prop.value)
                .split('\0')
                .nth(1)
                .unwrap_or("Unknown")
                .to_string();

            Ok(WindowMetadata {
                source_app,
                window_title,
                source_url: None,  // X11 无法直接获取浏览器 URL
            })
        }

        /// 获取 Wayland 活动窗口信息
        fn get_wayland_window_info(&self) -> Result<WindowMetadata, ClipboardError> {
            // Wayland 安全模型限制，无法获取其他应用窗口信息
            Ok(WindowMetadata {
                source_app: "Unknown".to_string(),
                window_title: "".to_string(),
                source_url: None,
            })
        }
    }
}
```

### 2.3 Windows 实现

```rust
// src-tauri/src/platform/windows.rs
#[cfg(target_os = "windows")]
pub mod windows {
    use super::*;
    use windows::Win32::Foundation::*;
    use windows::Win32::System::DataExchange::*;
    use windows::Win32::UI::WindowsAndMessaging::*;
    use std::sync::{Arc, Mutex};

    pub struct WindowsClipboard {
        hwnd: HWND,
        last_sequence: Arc<Mutex<u32>>,
    }

    impl WindowsClipboard {
        pub fn new() -> Result<Self, ClipboardError> {
            // 创建隐藏窗口用于接收剪切板消息
            unsafe {
                let hwnd = CreateWindowExW(
                    WINDOW_EX_STYLE(0),
                    w!("STATIC"),
                    w!("ClipBox Clipboard Listener"),
                    WS_OVERLAPPEDWINDOW,
                    0, 0, 0, 0,
                    None,
                    None,
                    None,
                    None,
                );

                if hwnd.0 == 0 {
                    return Err(ClipboardError::PlatformError(
                        "Failed to create window".to_string()
                    ));
                }

                Ok(Self {
                    hwnd,
                    last_sequence: Arc::new(Mutex::new(0)),
                })
            }
        }
    }

    #[async_trait]
    impl PlatformClipboard for WindowsClipboard {
        fn register_listener<F>(&self, callback: F) -> Result<(), ClipboardError>
        where
            F: Fn(ClipboardContent) -> BoxFuture<'static, Result<(), ClipboardError>> + Send + Sync + 'static,
        {
            unsafe {
                // 注册剪切板格式监听器
                if !AddClipboardFormatListener(self.hwnd).as_bool() {
                    return Err(ClipboardError::PlatformError(
                        "Failed to register clipboard listener".to_string()
                    ));
                }
            }

            // 启动消息循环
            let hwnd = self.hwnd;
            let last_sequence = self.last_sequence.clone();

            std::thread::spawn(move || {
                unsafe {
                    let mut msg = MSG::default();
                    while GetMessageW(&mut msg, hwnd, 0, 0).as_bool() {
                        if msg.message == WM_CLIPBOARDUPDATE {
                            // 获取剪切板序列号
                            let sequence = GetClipboardSequenceNumber();
                            let mut last = last_sequence.lock().unwrap();

                            if sequence != *last {
                                *last = sequence;
                                drop(last);

                                // 获取剪切板内容
                                if let Ok(content) = Self::get_clipboard_content_sync() {
                                    // 异步调用回调
                                    tokio::runtime::Runtime::new()
                                        .unwrap()
                                        .block_on(async {
                                            let _ = callback(content).await;
                                        });
                                }
                            }
                        }

                        TranslateMessage(&msg);
                        DispatchMessageW(&msg);
                    }
                }
            });

            Ok(())
        }

        async fn get_content(&self) -> Result<ClipboardContent, ClipboardError> {
            Self::get_clipboard_content_sync()
        }

        async fn set_content(&self, content: ClipboardContent) -> Result<(), ClipboardError> {
            use arboard::Clipboard;
            let mut clipboard = Clipboard::new()
                .map_err(|e| ClipboardError::AccessError(e.to_string()))?;

            match content {
                ClipboardContent::Text(text) => {
                    clipboard.set_text(text)
                        .map_err(|e| ClipboardError::AccessError(e.to_string()))?;
                }
                ClipboardContent::Image(data) => {
                    let img = image::load_from_memory(&data)
                        .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;
                    clipboard.set_image(arboard::ImageData {
                        width: img.width() as usize,
                        height: img.height() as usize,
                        bytes: img.to_rgba8().into_raw().into(),
                    })
                    .map_err(|e| ClipboardError::AccessError(e.to_string()))?;
                }
                _ => return Err(ClipboardError::UnsupportedType),
            }

            Ok(())
        }

        fn get_active_window_info(&self) -> Result<WindowMetadata, ClipboardError> {
            unsafe {
                let hwnd = GetForegroundWindow();
                if hwnd.0 == 0 {
                    return Ok(WindowMetadata {
                        source_app: "Unknown".to_string(),
                        window_title: "".to_string(),
                        source_url: None,
                    });
                }

                // 获取窗口标题
                let mut title = vec![0u16; 256];
                let len = GetWindowTextW(hwnd, &mut title);
                let window_title = String::from_utf16_lossy(&title[..len as usize]);

                // 获取进程名
                let mut process_id = 0;
                GetWindowThreadProcessId(hwnd, Some(&mut process_id));

                let source_app = Self::get_process_name(process_id)
                    .unwrap_or_else(|_| "Unknown".to_string());

                // 尝试获取浏览器 URL (仅 Chrome/Edge)
                let source_url = Self::try_get_browser_url(hwnd, &source_app);

                Ok(WindowMetadata {
                    source_app,
                    window_title,
                    source_url,
                })
            }
        }

        async fn clear(&self) -> Result<(), ClipboardError> {
            unsafe {
                if !OpenClipboard(self.hwnd).as_bool() {
                    return Err(ClipboardError::AccessError("Failed to open clipboard".to_string()));
                }
                EmptyClipboard();
                CloseClipboard();
            }
            Ok(())
        }
    }

    impl WindowsClipboard {
        fn get_clipboard_content_sync() -> Result<ClipboardContent, ClipboardError> {
            use arboard::Clipboard;
            let mut clipboard = Clipboard::new()
                .map_err(|e| ClipboardError::AccessError(e.to_string()))?;

            // 优先尝试文本
            if let Ok(text) = clipboard.get_text() {
                return Ok(ClipboardContent::Text(text));
            }

            // 尝试图片
            if let Ok(img) = clipboard.get_image() {
                let rgba = image::RgbaImage::from_raw(
                    img.width as u32,
                    img.height as u32,
                    img.bytes.to_vec(),
                ).ok_or(ClipboardError::PlatformError("Invalid image data".to_string()))?;

                let mut buffer = Vec::new();
                rgba.write_to(&mut std::io::Cursor::new(&mut buffer), image::ImageFormat::Png)
                    .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;

                return Ok(ClipboardContent::Image(buffer));
            }

            Err(ClipboardError::UnsupportedType)
        }

        fn get_process_name(process_id: u32) -> Result<String, ClipboardError> {
            use windows::Win32::System::ProcessStatus::*;
            use windows::Win32::System::Threading::*;

            unsafe {
                let handle = OpenProcess(PROCESS_QUERY_INFORMATION | PROCESS_VM_READ, false, process_id)
                    .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;

                let mut name = vec![0u16; 260];
                let mut size = name.len() as u32;

                if QueryFullProcessImageNameW(handle, PROCESS_NAME_WIN32, &mut name, &mut size).as_bool() {
                    let path = String::from_utf16_lossy(&name[..size as usize]);
                    let exe_name = std::path::Path::new(&path)
                        .file_name()
                        .and_then(|n| n.to_str())
                        .unwrap_or("Unknown")
                        .to_string();

                    CloseHandle(handle);
                    return Ok(exe_name);
                }

                CloseHandle(handle);
                Err(ClipboardError::PlatformError("Failed to get process name".to_string()))
            }
        }

        fn try_get_browser_url(hwnd: HWND, app_name: &str) -> Option<String> {
            // 仅支持 Chrome/Edge (通过 UI Automation)
            if !app_name.to_lowercase().contains("chrome") && !app_name.to_lowercase().contains("msedge") {
                return None;
            }

            // 简化实现：通过窗口标题提取 URL (不完美但可用)
            // 完整实现需要 UI Automation API
            None
        }
    }
}
```

```rust
// src-tauri/src/platform/macos.rs
#[cfg(target_os = "macos")]
pub mod macos {
    use super::*;
    use cocoa::appkit::{NSPasteboard, NSPasteboardTypeString, NSPasteboardTypePNG};
    use cocoa::base::{id, nil};
    use cocoa::foundation::{NSString, NSAutoreleasePool};
    use objc::{msg_send, sel, sel_impl};
    use std::sync::{Arc, Mutex};

    pub struct MacOSClipboard {
        pasteboard: id,
        last_change_count: Arc<Mutex<i64>>,
    }

    impl MacOSClipboard {
        pub fn new() -> Result<Self, ClipboardError> {
            unsafe {
                let pasteboard: id = msg_send![class!(NSPasteboard), generalPasteboard];
                if pasteboard == nil {
                    return Err(ClipboardError::PlatformError(
                        "Failed to get pasteboard".to_string()
                    ));
                }

                Ok(Self {
                    pasteboard,
                    last_change_count: Arc::new(Mutex::new(0)),
                })
            }
        }
    }

    #[async_trait]
    impl PlatformClipboard for MacOSClipboard {
        fn register_listener<F>(&self, callback: F) -> Result<(), ClipboardError>
        where
            F: Fn(ClipboardContent) -> BoxFuture<'static, Result<(), ClipboardError>> + Send + Sync + 'static,
        {
            let pasteboard = self.pasteboard;
            let last_change_count = self.last_change_count.clone();

            tokio::spawn(async move {
                loop {
                    unsafe {
                        let _pool = NSAutoreleasePool::new(nil);

                        let change_count: i64 = msg_send![pasteboard, changeCount];
                        let mut last = last_change_count.lock().unwrap();

                        if change_count != *last {
                            *last = change_count;
                            drop(last);

                            // 获取剪切板内容
                            if let Ok(content) = Self::get_pasteboard_content(pasteboard) {
                                let _ = callback(content).await;
                            }
                        }
                    }

                    tokio::time::sleep(tokio::time::Duration::from_millis(500)).await;
                }
            });

            Ok(())
        }

        async fn get_content(&self) -> Result<ClipboardContent, ClipboardError> {
            unsafe {
                Self::get_pasteboard_content(self.pasteboard)
            }
        }

        async fn set_content(&self, content: ClipboardContent) -> Result<(), ClipboardError> {
            use arboard::Clipboard;
            let mut clipboard = Clipboard::new()
                .map_err(|e| ClipboardError::AccessError(e.to_string()))?;

            match content {
                ClipboardContent::Text(text) => {
                    clipboard.set_text(text)
                        .map_err(|e| ClipboardError::AccessError(e.to_string()))?;
                }
                ClipboardContent::Image(data) => {
                    let img = image::load_from_memory(&data)
                        .map_err(|e| ClipboardError::PlatformError(e.to_string()))?;
                    clipboard.set_image(arboard::ImageData {
                        width: img.width() as usize,
                        height: img.height() as usize,
                        bytes: img.to_rgba8().into_raw().into(),
                    })
                    .map_err(|e| ClipboardError::AccessError(e.to_string()))?;
                }
                _ => return Err(ClipboardError::UnsupportedType),
            }

            Ok(())
        }

        fn get_active_window_info(&self) -> Result<WindowMetadata, ClipboardError> {
            unsafe {
                use cocoa::appkit::NSWorkspace;

                let workspace: id = msg_send![class!(NSWorkspace), sharedWorkspace];
                let active_app: id = msg_send![workspace, frontmostApplication];

                if active_app == nil {
                    return Ok(WindowMetadata {
                        source_app: "Unknown".to_string(),
                        window_title: "".to_string(),
                        source_url: None,
                    });
                }

                let app_name: id = msg_send![active_app, localizedName];
                let source_app = if app_name != nil {
                    let cstr: *const i8 = msg_send![app_name, UTF8String];
                    std::ffi::CStr::from_ptr(cstr).to_string_lossy().to_string()
                } else {
                    "Unknown".to_string()
                };

                // macOS 无法轻易获取窗口标题（需要 Accessibility API 权限）
                Ok(WindowMetadata {
                    source_app,
                    window_title: "".to_string(),
                    source_url: None,
                })
            }
        }

        async fn clear(&self) -> Result<(), ClipboardError> {
            unsafe {
                let _: () = msg_send![self.pasteboard, clearContents];
            }
            Ok(())
        }
    }

    impl MacOSClipboard {
        unsafe fn get_pasteboard_content(pasteboard: id) -> Result<ClipboardContent, ClipboardError> {
            // 尝试获取文本
            let string_type = NSString::alloc(nil).init_str(NSPasteboardTypeString);
            let text: id = msg_send![pasteboard, stringForType: string_type];

            if text != nil {
                let cstr: *const i8 = msg_send![text, UTF8String];
                let content = std::ffi::CStr::from_ptr(cstr).to_string_lossy().to_string();
                return Ok(ClipboardContent::Text(content));
            }

            // 尝试获取图片
            let png_type = NSString::alloc(nil).init_str(NSPasteboardTypePNG);
            let image_data: id = msg_send![pasteboard, dataForType: png_type];

            if image_data != nil {
                let length: usize = msg_send![image_data, length];
                let bytes: *const u8 = msg_send![image_data, bytes];
                let data = std::slice::from_raw_parts(bytes, length).to_vec();
                return Ok(ClipboardContent::Image(data));
            }

            Err(ClipboardError::UnsupportedType)
        }
    }
}
```

## 3. 跨平台总结

### 3.1 平台差异对比表

| 功能 | Linux (X11) | Linux (Wayland) | Windows | macOS |
|------|-------------|-----------------|---------|-------|
| 剪切板监听 | 轮询 (500ms) | 轮询 (500ms) | 原生事件 (WM_CLIPBOARDUPDATE) | 轮询 (changeCount) |
| 获取窗口标题 | ✅ 支持 | ❌ 受限 | ✅ 支持 | ❌ 需权限 |
| 获取应用名 | ⚠️ 部分支持 | ❌ 受限 | ✅ 支持 | ✅ 支持 |
| 获取浏览器 URL | ❌ 不支持 | ❌ 不支持 | ⚠️ 部分支持 | ❌ 不支持 |
| 图片剪切板 | ✅ 支持 | ✅ 支持 | ✅ 支持 | ✅ 支持 |

### 3.2 跨平台最佳实践

1. **使用 Trait 抽象**: 所有平台特定代码通过 `PlatformClipboard` trait 统一接口
2. **条件编译**: 使用 `#[cfg(target_os = "...")]` 确保只编译目标平台代码
3. **降级方案**: 当平台不支持某功能时，提供降级实现（如轮询代替事件）
4. **统一依赖**: 优先使用跨平台 crate（如 `arboard`、`image`）
5. **测试覆盖**: 在所有目标平台上进行集成测试
